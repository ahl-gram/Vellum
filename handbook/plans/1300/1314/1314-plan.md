```ts
import type { Rng } from "./rng.ts";

export type FreshPick = (list: readonly string[]) => string;

export function freshPicker(rng: Rng): FreshPick {
  const used = new Map<readonly string[], Set<string>>();
  return (list) => {
    let seen = used.get(list);
    if (!seen) {
      seen = new Set();
      used.set(list, seen);
    }
    if (seen.size >= list.length) seen.clear();
    const choice = rng.pick(list.filter((x) => !seen.has(x)));
    seen.add(choice);
    return choice;
  };
}
```

```ts
import { createHash } from "node:crypto";
import { createRng } from "../../src/core/rng.ts";
import { defaultRecipe, generateWorld } from "../../src/world/generate.ts";
import { createLoreWriter } from "../../src/society/lore.ts";
import { buildVoyageLog, type VoyageLogPort } from "../../src/world/voyage-log.ts";
import { buildPlaceManifest } from "../../src/render/place-manifest.ts";
import { buildVoyagePlan } from "../../src/render/voyage.ts";
import { buildSurvey } from "../../src/render/survey.ts";
import { routeVoyage } from "../../src/render/voyage-route.ts";
import { defaultJourney } from "../../src/itinerary/route.ts";
import { buildRibbonInput } from "../../src/itinerary/input.ts";
import { LEXICON } from "../../src/world/founding/lexicon.ts";

const SEEDS = [42, 7, 100, 953, 2024];

const gridLen = (pts: ReadonlyArray<{ x: number; y: number }>): number => {
  let s = 0;
  for (let i = 1; i < pts.length; i++) s += Math.hypot(pts[i]!.x - pts[i - 1]!.x, pts[i]!.y - pts[i - 1]!.y);
  return s;
};

function voyageOf(seed: number, world: ReturnType<typeof generateWorld>): unknown {
  const manifest = buildPlaceManifest(world, 1500);
  const plan = buildVoyagePlan(manifest.places, manifest.presentYear);
  const survey = buildSurvey(world.elev, world.seaLevel, world.roads);
  const sites = manifest.places.map((p) => ({ idx: p.idx, x: p.gx, y: p.gy }));
  const routed = routeVoyage(plan.legs, sites, survey);
  if (routed.length === 0) return null;
  const byIdx = new Map(manifest.places.map((p) => [p.idx, p]));
  const ports: VoyageLogPort[] = plan.ports.map((pt, i) => {
    const pm = byIdx.get(pt.idx)!;
    return {
      idx: pm.idx,
      name: pm.name,
      kind: pm.kind,
      founded: pm.founded,
      arrivalMode: i === 0 ? null : routed[i - 1]!.mode,
      inlandHandoff: i === 0 ? false : routed[i - 1]!.inlandHandoff,
      legLength: i === 0 ? 0 : gridLen(routed[i - 1]!.points),
    };
  });
  const closing = routed[routed.length - 1]!;
  return buildVoyageLog(ports, manifest.presentYear, seed, world.title.subtitle, {
    arrivalMode: closing.mode,
    inlandHandoff: closing.inlandHandoff,
    legLength: gridLen(closing.points),
  });
}

const out: Record<string, unknown> = { lexicon: LEXICON };
for (const seed of SEEDS) {
  const world = generateWorld(defaultRecipe(seed));
  const lore = createLoreWriter(world, createRng(seed).fork("lore"));
  const journey = defaultJourney(world);
  out[`seed ${seed}`] = {
    history: world.history.events,
    beasts: world.beasts,
    settlementNotes: world.settlements.map((s) => lore.settlementNote(s)),
    realmNotes: world.names.realms.map((r) => lore.realmNote(r)),
    voyage: voyageOf(seed, world),
    ribbon: journey === null ? null : buildRibbonInput(world, journey.fromIdx, journey.toIdx),
  };
}
const text = JSON.stringify(out);
const sum = (k: string): string => createHash("sha256").update(JSON.stringify(out[k])).digest("hex").slice(0, 16);
console.error(`bytes ${text.length}; whole ${createHash("sha256").update(text).digest("hex")}`);
for (const k of Object.keys(out)) console.error(`${k} ${sum(k)}`);
process.stdout.write(text);
```
