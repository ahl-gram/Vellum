## The design

1. **One role body per agent, under `handbook/agents/roles/<name>.md`**: the text after today's front matter, moved byte for byte (commit 1), then with each of 17 Claude-only passages replaced by a `{{harness:<slot>}}` marker (commit 3).
2. **Two adapters, `handbook/agents/adapters/claude-code.md` and `codex-local.md`.** Each opens with a preamble the generator never emits, then gives each slot under a comment line `<!-- slot: <name> -->` as exactly one line of text, and may end with an appendix after a comment line `<!-- appendix -->` that is appended to every role of that harness. **The Claude Code slots are the passages' own words**, so every Claude definition stays byte for byte what it is today (ruled: fill-in slots). The Codex slots are written out in full under Files, and the Codex appendix says how a role reads what Claude Code gives a session (the scratchpad, Bash, dispatching, the read and write limits, the decision menu). Claude Code's adapter has no appendix.
3. **A typed manifest, `ROLES` in `scripts/agent-infra/roles.ts`**: each role's name, description, Claude tool list, isolation and color, copied from today's front matter; both harnesses share the description.
4. **A model policy, `.agents/model-policy.json`, read as `unknown` and narrowed by `parseModelPolicy`**: per harness, the approved models with the efforts each allows (the client's own list), and per role the model and effort. A missing harness, role, model or effort, an extra key, an unapproved model and an effort that model does not allow are each refused; `resolveModel` never falls back. Claude Code's column is today's front matter; Codex's is Alex's D8 ruling.
5. **The generator, `scripts/agent-infra/generate.ts`**: `planGeneratedFiles` reads the policy, the bodies and both adapters, refuses an adapter slot no body uses and a marker no adapter defines, and renders each role for each harness; `checkGeneratedFiles` compares every planned file with the one on disk and reports every tracked file under `.claude/agents/` or `.codex/agents/` no role generates, writing nothing; `writeGeneratedFiles` refuses, before writing anything, when a planned path is a link or a folder, writes only the files that differ, and never deletes. `npm run agents:generate` and `npm run agents:check` run it from `scripts/agent-infra/main.ts`, in the folder npm runs them in, the repository root.
6. **The renderers, `scripts/agent-infra/render.ts`**: Claude Code's front matter is `name`, `description`, `tools` when the role has one, `model`, `effort`, `isolation` when the role has one, `color`, as today's files have it, then a blank line and the body. Codex's file is five keys, in Sub 1's fixture order: `name`, `description`, `model`, `model_reasoning_effort` as basic strings, and `developer_instructions` as a multi-line literal string holding the filled body and the appendix.
7. **Tests**: three new files under `test/repo/` (the live tree, the pieces on fixtures, whole trees in temporary repositories), and the follow-the-move edits to six existing ones. `npm test` runs the check through AD1, so CI runs it with no vendor account.
8. **CI classification (D12, ruled)**: `.codex/agents/` joins `.claude/agents/` as documentation in `scripts/docs-only.ts`; `.agents/model-policy.json` and `scripts/agent-infra/` stay code, so a model change always runs every lane and a reworded role runs the unit suite, which holds the drift check, as every diff does.
9. **Doctrine**: one new paragraph in `handbook/specs/conventions.md`, and edits to that file, `handbook/specs/orchestration.md`, `handbook/specs/rulebook.md` and `handbook/specs/development-workflow.md` (Rosters and doctrine).

- **Writing `.claude/settings.json`**: nothing here needs a setting; AT2 holds that it survives untouched.

- **Leaving `.claude/agents` out of the memory-pointers and prose-paths roots once `handbook/agents` is in**: the descriptions live in `scripts/agent-infra/roles.ts` and reach prose only through the generated front matter.

import { join } from "node:path";
import { GenerationError, messageOf } from "./errors.ts";
import { HARNESSES, POLICY_FILE, type HarnessId, type RoleDefinition } from "./roles.ts";

export const EFFORTS = ["low", "medium", "high", "xhigh", "max", "ultra"] as const;
export type Effort = (typeof EFFORTS)[number];
export type ModelSelection = { readonly model: string; readonly effort: Effort };
export type HarnessPolicy = {
  readonly models: ReadonlyMap<string, readonly Effort[]>;
  readonly roles: ReadonlyMap<string, ModelSelection>;
};
export type ModelPolicy = ReadonlyMap<HarnessId, HarnessPolicy>;

type Table = Readonly<Record<string, unknown>>;

const isTable = (value: unknown): value is Table =>
  typeof value === "object" && value !== null && !Array.isArray(value);
const isEffort = (value: unknown): value is Effort => EFFORTS.some((effort) => effort === value);

function exactKeys(value: unknown, keys: readonly string[], where: string): Table {
  if (!isTable(value)) throw new GenerationError(`${where} is not an object`);
  const extra = Object.keys(value).filter((key) => !keys.includes(key));
  if (extra.length > 0) throw new GenerationError(`${where} holds a key it may not: ${extra.join(", ")}`);
  const missing = keys.filter((key) => !Object.hasOwn(value, key));
  if (missing.length > 0) throw new GenerationError(`${where} has no ${missing.join(", ")}`);
  return value;
}

function effortList(value: unknown): readonly Effort[] | null {
  if (!Array.isArray(value)) return null;
  const items: readonly unknown[] = value;
  return items.length > 0 && items.every(isEffort) && new Set(items).size === items.length ? items : null;
}

function parseModels(value: unknown, where: string): ReadonlyMap<string, readonly Effort[]> {
  if (!isTable(value) || Object.keys(value).length === 0) throw new GenerationError(`${where} approves no model`);
  return new Map(
    Object.entries(value).map(([model, efforts]) => {
      const list = effortList(efforts);
      if (list === null)
        throw new GenerationError(`${where} ${model} is not a list of distinct efforts from ${EFFORTS.join(", ")}`);
      return [model, list];
    }),
  );
}

function parseRoles(
  value: unknown,
  models: ReadonlyMap<string, readonly Effort[]>,
  names: readonly string[],
  where: string,
): ReadonlyMap<string, ModelSelection> {
  const table = exactKeys(value, names, where);
  return new Map(
    names.map((name) => {
      const { model, effort } = exactKeys(table[name], ["model", "effort"], `${where} ${name}`);
      if (typeof model !== "string" || !models.has(model))
        throw new GenerationError(
          `${where} ${name} names a model the policy does not approve: ${JSON.stringify(model)}`,
        );
      if (!isEffort(effort) || !(models.get(model) ?? []).includes(effort))
        throw new GenerationError(
          `${where} ${name} asks ${model} for an effort the policy does not approve for it: ${JSON.stringify(effort)}`,
        );
      return [name, { model, effort }];
    }),
  );
}

export function parseModelPolicy(raw: unknown, roles: readonly RoleDefinition[]): ModelPolicy {
  const top = exactKeys(raw, HARNESSES, POLICY_FILE);
  const names = roles.map((role) => role.name);
  return new Map(
    HARNESSES.map((harness) => {
      const where = `${POLICY_FILE} ${harness}`;
      const section = exactKeys(top[harness], ["models", "roles"], where);
      const models = parseModels(section["models"], `${where} models`);
      return [harness, { models, roles: parseRoles(section["roles"], models, names, `${where} roles`) }];
    }),
  );
}

export function loadModelPolicy(root: string, roles: readonly RoleDefinition[]): ModelPolicy {
  let raw: unknown;
  try {
    raw = JSON.parse(readFileSync(join(root, POLICY_FILE), "utf8"));
  } catch (error) {
    throw new GenerationError(`${POLICY_FILE} cannot be read as JSON: ${messageOf(error)}`);
  }
  return parseModelPolicy(raw, roles);
}

export function resolveModel(policy: ModelPolicy, harness: HarnessId, role: string): ModelSelection {
  const selection = policy.get(harness)?.roles.get(role);
  if (selection === undefined)
    throw new GenerationError(`${POLICY_FILE} gives ${role} no model under ${harness}, and a role never inherits one`);
  return selection;
}

import { GenerationError } from "./errors.ts";

export type Adapter = { readonly slots: ReadonlyMap<string, string>; readonly appendix: string };

const SLOT_LINE = /^<!-- slot: ([a-z0-9]+(?:-[a-z0-9]+)*) -->$/;
const APPENDIX_LINE = "<!-- appendix -->";
const MARKER = /\{\{harness:([a-z0-9]+(?:-[a-z0-9]+)*)\}\}/g;

type Section = { readonly marker: string; readonly lines: string[] };

function sectionsOf(text: string): readonly Section[] {
  const sections: Section[] = [{ marker: "", lines: [] }];
  for (const line of text.split("\n")) {
    if (SLOT_LINE.test(line) || line === APPENDIX_LINE) sections.push({ marker: line, lines: [] });
    else sections.at(-1)?.lines.push(line);
  }
  return sections.slice(1);
}

function withoutEdgeBlanks(lines: readonly string[]): readonly string[] {
  const first = lines.findIndex((line) => line.trim() !== "");
  return first === -1 ? [] : lines.slice(first, lines.findLastIndex((line) => line.trim() !== "") + 1);
}

export function parseAdapter(text: string, where: string): Adapter {
  const sections = sectionsOf(text);
  const slots = new Map<string, string>();
  let appendix = "";
  sections.forEach(({ marker, lines }, at) => {
    const kept = withoutEdgeBlanks(lines);
    if (marker === APPENDIX_LINE) {
      if (at !== sections.length - 1) throw new GenerationError(`${where}: the appendix must be its last section`);
      appendix = kept.length === 0 ? "" : `\n${kept.join("\n")}\n`;
      return;
    }
    const name = SLOT_LINE.exec(marker)?.[1] ?? marker;
    if (slots.has(name)) throw new GenerationError(`${where} defines the slot ${name} twice`);
    const [only] = kept;
    if (kept.length !== 1 || only === undefined)
      throw new GenerationError(`${where}: the slot ${name} must be one line of text, and is ${kept.length}`);
    slots.set(name, only);
  });
  return { slots, appendix };
}

export const slotsUsedIn = (body: string): readonly string[] => [...body.matchAll(MARKER)].map((m) => m[1] ?? "");

export function fillSlots(body: string, adapter: Adapter, where: string, adapterWhere: string): string {
  const filled = body.replace(MARKER, (_marker: string, name: string) => {
    const text = adapter.slots.get(name);
    if (text === undefined)
      throw new GenerationError(`${where} uses the slot ${name}, which ${adapterWhere} does not define`);
    return text;
  });
  if (filled.includes("{{") || filled.includes("}}"))
    throw new GenerationError(`${where} holds a {{ or }} that is no {{harness:<name>}} slot, here or in a slot's text`);
  return filled;
}

const TAB = 0x09;
const LINE_FEED = 0x0a;

function controlIn(text: string, allowed: readonly number[]): string | null {
  for (const character of text) {
    const code = character.codePointAt(0) ?? 0;
    if ((code < 0x20 || code === 0x7f) && !allowed.includes(code))
      return `U+${code.toString(16).toUpperCase().padStart(4, "0")}`;
  }
  return null;
}

const excerpt = (text: string): string => JSON.stringify(text.slice(0, 60));

export function tomlBasicString(text: string): string {
  const control = controlIn(text, [TAB]);
  if (control !== null) throw new GenerationError(`a one-line TOML string cannot hold ${control}: ${excerpt(text)}`);
  const escaped = text.replaceAll("\\", "\\\\").replaceAll('"', '\\"');
  return `"${escaped}"`;
}

export function tomlMultilineLiteral(text: string): string {
  if (text.includes("'''")) throw new GenerationError(`a TOML literal string cannot hold ''': ${excerpt(text)}`);
  const control = controlIn(text, [TAB, LINE_FEED]);
  if (control !== null) throw new GenerationError(`a TOML literal string cannot hold ${control}: ${excerpt(text)}`);
  return `'''\n${text}'''`;
}

export function renderClaudeAgent(

}

import { fillSlots, parseAdapter, slotsUsedIn } from "./adapters.ts";
import { GenerationError, messageOf } from "./errors.ts";
import { loadModelPolicy, resolveModel } from "./model-policy.ts";
import { renderClaudeAgent, renderCodexAgent } from "./render.ts";
import {
  HARNESSES,
  OUTPUT_DIRS,
  POLICY_FILE,
  ROLES,
  adapterPath,
  outputPath,
  roleBodyPath,
  type HarnessId,
  type RoleDefinition,
} from "./roles.ts";

export type GeneratedFile = { readonly path: string; readonly text: string };
export type GenerationProblem = { readonly path: string; readonly problem: string };

const GENERATE = "npm run agents:generate";
// 2026-11-18: this git ls-files answers in under 10 ms on a Mac; thirty seconds is a cap on a hang, not a budget.
const GIT_TIMEOUT_MS = 30_000;
const RENDER: Readonly<Record<HarnessId, typeof renderClaudeAgent>> = {
  "claude-code": renderClaudeAgent,
  "codex-local": renderCodexAgent,
};

const byPath = (a: { readonly path: string }, b: { readonly path: string }): number =>
  a.path < b.path ? -1 : a.path > b.path ? 1 : 0;

function readInput(root: string, path: string): string {
  try {
    return readFileSync(join(root, path), "utf8");
  } catch (error) {
    throw new GenerationError(`${path} cannot be read: ${messageOf(error)}`);
  }
}

export function planGeneratedFiles(root: string, roles: readonly RoleDefinition[] = ROLES): readonly GeneratedFile[] {
  const policy = loadModelPolicy(root, roles);
  const bodies = roles.map((role) => [role, readInput(root, roleBodyPath(role.name))] as const);
  const used = new Set(bodies.flatMap(([, body]) => slotsUsedIn(body)));
  const files = HARNESSES.flatMap((harness) => {
    const adapter = parseAdapter(readInput(root, adapterPath(harness)), adapterPath(harness));
    const unused = [...adapter.slots.keys()].filter((name) => !used.has(name));
    if (unused.length > 0)
      throw new GenerationError(`${adapterPath(harness)} defines slots no role body uses: ${unused.join(", ")}`);
    return bodies.map(([role, body]) => ({
      path: outputPath(harness, role.name),
      text: RENDER[harness](
        role,
        resolveModel(policy, harness, role.name),
        fillSlots(body, adapter, roleBodyPath(role.name), adapterPath(harness)),
        adapter.appendix,
      ),
    }));
  });
  const paths = files.map((file) => file.path);
  const repeated = paths.filter((path, at) => paths.indexOf(path) !== at);
  if (repeated.length > 0) throw new GenerationError(`two roles would write ${repeated.join(", ")}`);
  return [...files].sort(byPath);
}

function trackedOutputs(root: string): readonly string[] {
  const listing = spawnSync("git", ["ls-files", "-z", "--", ...HARNESSES.map((h) => OUTPUT_DIRS[h])], {
    cwd: root,
    encoding: "utf8",
    timeout: GIT_TIMEOUT_MS,
  });
  if (listing.status !== 0)
    throw new GenerationError(
      `git ls-files failed, so the tracked definitions are unread: ${listing.error?.message ?? listing.stderr}`,
    );
  return listing.stdout.split("\0").filter(Boolean);
}

function driftOf(root: string, { path, text }: GeneratedFile): readonly GenerationProblem[] {
  const stat = lstatSync(join(root, path), { throwIfNoEntry: false });
  if (stat === undefined) return [{ path, problem: `is missing: run ${GENERATE}` }];
  if (!stat.isFile()) return [{ path, problem: "is not a regular file, so it is not the generated definition" }];
  if (readFileSync(join(root, path), "utf8") === text) return [];
  return [{ path, problem: `differs from what ${GENERATE} writes from handbook/agents/ and ${POLICY_FILE}` }];
}

export function checkGeneratedFiles(
  root: string,
  roles: readonly RoleDefinition[] = ROLES,
): readonly GenerationProblem[] {
  let planned: readonly GeneratedFile[];
  let tracked: readonly string[];
  try {
    planned = planGeneratedFiles(root, roles);
    tracked = trackedOutputs(root);
  } catch (error) {
    return [{ path: "scripts/agent-infra", problem: messageOf(error) }];
  }
  const plannedPaths = new Set(planned.map((file) => file.path));
  const strays = tracked
    .filter((path) => !plannedPaths.has(path))
    .map((path) => ({ path, problem: "is tracked, but no role in scripts/agent-infra/roles.ts generates it" }));
  return [...planned.flatMap((file) => driftOf(root, file)), ...strays].sort(byPath);
}

export function writeGeneratedFiles(root: string, roles: readonly RoleDefinition[] = ROLES): readonly string[] {
  const planned = planGeneratedFiles(root, roles);
  const blocked = planned.filter(
    ({ path }) => lstatSync(join(root, path), { throwIfNoEntry: false })?.isFile() === false,
  );
  if (blocked.length > 0)
    throw new GenerationError(
      `nothing written: ${blocked.map((file) => file.path).join(", ")} is not a regular file, so writing would go through it`,
    );
  const changed = planned.filter((file) => driftOf(root, file).length > 0);
  for (const { path, text } of changed) {
    mkdirSync(dirname(join(root, path)), { recursive: true });
    writeFileSync(join(root, path), text);
  }
  return changed.map((file) => file.path);
}
