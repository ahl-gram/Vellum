## The README path check, `1427-readme-paths.ts`

It is the first Primer archive's check (`handbook/plans/1200/1247/1247-plan.md`), its usage line renamed. It checks every backticked token in a README that looks like a path (a slash, a star or a file extension): a plain path must exist from the archive directory or from the repository root, a bare name must be the name of a file in the archive, and a pattern must match at least one file. Left-out files are named in the READMEs without backticks for this reason.

```ts
import { existsSync, readFileSync, readdirSync, statSync } from "node:fs";
import { basename, join, relative, resolve } from "node:path";

const [readmeArg, archiveArg, rootArg] = process.argv.slice(2);
if (!readmeArg || !archiveArg || !rootArg) throw new Error("usage: node 1427-readme-paths.ts <README.md> <archive dir> <repo root>");
const archive = resolve(archiveArg);
const root = resolve(rootArg);
const files = readdirSync(archive, { recursive: true, encoding: "utf8" })
  .filter((f) => statSync(join(archive, f)).isFile())
  .map((f) => f.split("\\").join("/"));
const tokens = [...readFileSync(readmeArg, "utf8").matchAll(/`([^`\s]+)`/g)].map((m) => m[1]!);
const pathLike = tokens.filter((t) => t.includes("/") || t.includes("*") || /\.[A-Za-z0-9]+$/.test(t));
const globRe = (g: string): RegExp => new RegExp("^" + g.split("*").map((s) => s.replace(/[.+?^${}()|[\]\\]/g, "\\$&")).join("[^/]*") + "$");
const resolves = (t: string): boolean => {
  if (t.includes("*")) {
    const re = globRe(t);
    return files.some((f) => re.test(f) || re.test(basename(f)));
  }
  if (existsSync(resolve(archive, t)) || existsSync(resolve(root, t))) return true;
  return !t.includes("/") && files.some((f) => basename(f) === t);
};
const unresolved = [...new Set(pathLike)].filter((t) => !resolves(t));
for (const t of unresolved) console.log(`UNRESOLVED: \`${t}\``);
console.log(`${pathLike.length} backticked paths (${new Set(pathLike).size} distinct), ${unresolved.length} unresolved, ${files.length} files under ${relative(root, archive)}`);
process.exitCode = unresolved.length === 0 ? 0 : 1;
```

import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";

const PLAN = process.argv[2] ?? "handbook/plans/1400/1427/1427-plan.md";
const WANT: Record<string, string> = {
  "design/the-primer-seats/README.md": "15aa7a92f9e8475260e93dcb64cb8c442f02dcbb0f5e3fe5607834cde702fd1f",

};
const lines = readFileSync(PLAN, "utf8").split("\n");
let fail = false;
for (const [path, want] of Object.entries(WANT)) {
  const head = lines.indexOf(`### \`${path}\`, verbatim`);
  if (head < 0) throw new Error(`STOP: no heading for ${path} in ${PLAN}`);
  const open = lines.indexOf("````markdown", head);
  const close = open < 0 ? -1 : lines.indexOf("````", open + 1);
  if (open < 0 || close < 0) throw new Error(`STOP: no fenced block for ${path} in ${PLAN}`);
  const text = lines.slice(open + 1, close).join("\n") + "\n";
  writeFileSync(path, text);
  const sum = createHash("sha256").update(text).digest("hex");
  const ok = sum === want;
  if (!ok) fail = true;
  console.log(`${path}: ${text.split("\n").length - 1} lines, sha256 ${sum}${ok ? "" : `, STOP: the plan's sum is ${want}`}`);
}
console.log(fail ? "READMES DIFFER" : "READMES OK");
process.exitCode = fail ? 1 : 0;
```
