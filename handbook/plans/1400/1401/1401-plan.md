import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { ghRefusal, requiredHeadings } from "../../scripts/agent-infra/hooks/pr-body.ts";

const DIR = import.meta.dirname;
const TEMPLATE = process.argv[2] ?? join(DIR, "..", "..", ".github", "PULL_REQUEST_TEMPLATE.md");
const SECTIONS = ["## Guards", "## Ran", "## Records", "## Rulings", "## Look for these when you use it"];
const SLIP = [
  "<!-- The closing slip: CLAUDE.md, Process. -->",
  "*Drafted by*",
  "**<Claude Code or Codex> <version>**",
  "*the pages above, and the session that set them down: <session link or codex session id>*",
];
const body = (sections: readonly string[], slip: readonly string[]): string =>
  [
    "#1401 `Attribution for both harnesses`",
    "",
    "Closes #1401",
    "",
    "A fixture paragraph.",
    "",
    ...sections.flatMap((s) => [s, "", "None.", ""]),
    ...slip,
    "",
  ].join("\n");
const CLAUDE_SLIP = [
  "*Drafted by*",
  "**Claude Code 2.1.328**",
  "*the pages above, and the session that set them down: https://claude.ai/code/session_01DktzEvBu42UVq6brZ4Ea9h*",
];
const CODEX_SLIP = [
  "*Drafted by*",
  "**Codex 0.170.0**",
  "*the pages above, and the session that set them down: codex session 01a1f513-8c03-7e52-b1c2-5a3db05df020*",
];
const FIXTURES: readonly (readonly [string, string, string | null])[] = [
  ["1401-body-claude.md", body(SECTIONS, CLAUDE_SLIP), null],
  ["1401-body-codex.md", body(SECTIONS, CODEX_SLIP), null],
  ["1401-body-no-rulings.md", body(SECTIONS.filter((s) => s !== "## Rulings"), CLAUDE_SLIP), 'skips "## Rulings"'],
];

const failures: string[] = [];
const text = readFileSync(TEMPLATE, "utf8");
const endsWithSlip = text.endsWith(`\n\n${SLIP.join("\n")}\n`);
console.log(`template ${TEMPLATE}: ends with the ruled slip after one blank line: ${endsWithSlip}`);
if (!endsWithSlip) failures.push("the template does not end with the ruled slip after one blank line");
const headings = requiredHeadings(TEMPLATE);
console.log(`headings: ${JSON.stringify(headings)}`);
if (JSON.stringify(headings) !== JSON.stringify(SECTIONS)) failures.push("the template's sections are not the five");
for (const [name, fixture, expectDeny] of FIXTURES) {
  writeFileSync(join(DIR, name), fixture);
  const command = `gh pr create --title "#1401 probe" --body-file ${name}`;
  const [deny, warn] = ghRefusal(command, command, DIR, TEMPLATE);
  console.log(`${name}: deny=${deny === null ? "null" : JSON.stringify(deny.slice(0, 60))} warn=${warn === null ? "null" : JSON.stringify(warn.slice(0, 60))}`);
  const denyOk = expectDeny === null ? deny === null : deny !== null && deny.includes(expectDeny);
  if (!denyOk || warn !== null) failures.push(`${name}: expected deny ${expectDeny ?? "null"} and no warning`);
}
for (const failure of failures) console.log(`FAIL ${failure}`);
console.log(failures.length ? `${failures.length} failed` : "ok, every reading as expected");
process.exit(failures.length ? 1 : 0);
