## The README path check, `1247-readme-paths.ts`

```ts
import { existsSync, readFileSync, readdirSync, statSync } from "node:fs";
import { basename, join, relative, resolve } from "node:path";

const [readmeArg, archiveArg, rootArg] = process.argv.slice(2);
if (!readmeArg || !archiveArg || !rootArg) throw new Error("usage: node 1247-readme-paths.ts <README.md> <archive dir> <repo root>");
const archive = resolve(archiveArg);
const root = resolve(rootArg);
const files = readdirSync(archive, { recursive: true, encoding: "utf8" })
  .filter((f) => statSync(join(archive, f)).isFile())
  .map((f) => f.split("\\").join("/"));
const tokens = [...readFileSync(readmeArg, "utf8").matchAll(/`([^`\s]+)`/g)].map((m) => m[1]!);
const pathLike = tokens.filter((t) => t.includes("/") || t.includes("*") || /\.[A-Za-z0-9]+$/.test(t));
const globRe = (g: string): RegExp => new RegExp("^" + g.split("*").map((s) => s.replace(/[.+?^${}()|[\]\\]/g, "\\$&")).join("[^/]*") + "$");
const resolves = (t: string): boolean => {
  if (t.includes("*")) {
    const re = globRe(t);
    return files.some((f) => re.test(f) || re.test(basename(f)));
  }
  if (existsSync(resolve(archive, t)) || existsSync(resolve(root, t))) return true;
  return !t.includes("/") && files.some((f) => basename(f) === t);
};
const unresolved = [...new Set(pathLike)].filter((t) => !resolves(t));
for (const t of unresolved) console.log(`UNRESOLVED: \`${t}\``);
console.log(`${pathLike.length} backticked paths (${new Set(pathLike).size} distinct), ${unresolved.length} unresolved, ${files.length} files under ${relative(root, archive)}`);
process.exitCode = unresolved.length === 0 ? 0 : 1;
```

It checks every backticked token in the README that looks like a path (a slash, a star or a file extension): a plain path must exist from `design/the-primer/` or from the repository root, a bare name must be the name of a file in the archive, and a pattern must match at least one file. Proved at planning to red on a still that does not exist, a pattern that matches nothing, and each left-out path written in backticks (`shots/landings/`, `sheets/sheet-6-the-name.png`, `refs/images/manicule.jpg`): 5 of 5 reported UNRESOLVED, exit 1. Left-out files are named in the README without backticks for this reason.
