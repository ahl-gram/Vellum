## The design

### The shape (`src/site/shared/primer/shape.ts`)

A ladder per room keyed by its route; lessons of steps; each step's sentence (`act`) with its controls' own words in braces; `words`, the controls the sentence marks, each saying where the control is found (`css`) and how its words are read (`read`); `relies`, the controls the step leans on that its sentence does not mark (ruling A1); `at`, the address its Try it opens; `first`, what the reader has done before the controls are read, for whatever the address cannot carry; `do`, the step itself; `then`, what shows once it is done. `kind` marks a play step or a commission; `only` marks a tablet step or a motion step (ruling 15).

The shape carries everything all nine of the round's ladders use, not only the Explorer's, because Sub 6 (Issue #1252) owns `src/site/shared/primer/<room>.ts` and the list in `ladders.ts`, and not `shape.ts`: so `entered` (a subroom's way in), `read: "title"` and `read: "placeholder"`, `state: "hidden"` (the Hunt's presses before a win), `only: "motion"` (the Reading Room's pace step), and the act fields `choose`, `nth` and `at` that the other rooms' steps need (the mapping is the Appendix) are all here from the start. Of these, the Explorer's data uses `read: "placeholder"`, `choose`, and `only: "touch"`; the rest have their first data in Sub 6.

**An act is declarative.** The round's `eval` acts (JavaScript strings calling the round harness's own `tick()` and `until()`) and its `ticks` acts (a count of 100 ms sleeps) do not ship. In their place: `click`, `set`, `choose`, `key`, `pick`, `repeat`, `wait`. Each says what the reader does; Sub 4's driver (Issue #1250) performs it with real input through `e2e/support/input.ts` (`vellum-footguns` Gate 2 item 1: a gesture check drives the gesture, never an in-page event) and waits on a state, never the clock (Gate 2 item 6; `handbook/specs/settle-doctrine.md`).

**Dropped from the round's shape:** `Words.from` (the issue: the check measures it); `Words.constant` (the import of the room's own exported words is the link, ruling 10, so the documentation string goes); `Step.provisional` (no surface reads it, the round's own `ladder.md` says the mark is "never for the reader", and the provisional lists' one home is `handbook/specs/ui-design.md`; ruling 14 works through Sub 4's red, not through a field).

### Created: `src/site/shared/primer/shape.ts` (66 lines before its three comments)

Write it exactly as below.

```ts
// How a control's words are read: text, its own text; aria, its aria-label; label, the text of the label it sits in; option, a list offering an option in these words; road, a road press's verb and room joined by a space; attr, the attribute attr names; title and placeholder, those attributes; match starts or contains loosens the match; state held is a control standing in the held box, hidden one in the page but not shown until the step's own condition.
export interface Words {
  readonly css: string;
  readonly says: string;
  readonly read: "text" | "aria" | "label" | "option" | "road" | "attr" | "title" | "placeholder";
  readonly attr?: string;
  readonly match?: "starts" | "contains";
  readonly state?: "held" | "hidden";
}

export interface Spot {
  readonly x: number;
  readonly y: number;
}

// What the reader does, as a driver performs it with real input, never a script and never a wait on the clock: set with release false leaves the value uncommitted, choose takes the option in those words, repeat presses css until an element matching until is on the page, pick types text in the search box css and takes the entry named name, wait holds until an element matching css is on the page, nth counts matches from 0, and at is a point of the control's box in fractions of its width and height.
export type Act =
  | { readonly kind: "click"; readonly css: string; readonly nth?: number; readonly at?: Spot }
  | { readonly kind: "set"; readonly css: string; readonly value: string; readonly release?: false }
  | { readonly kind: "choose"; readonly css: string; readonly option: string }
  | { readonly kind: "key"; readonly css: string; readonly key: string }
  | { readonly kind: "pick"; readonly css: string; readonly text: string; readonly name: string }
  | { readonly kind: "repeat"; readonly css: string; readonly until: string }
  | { readonly kind: "wait"; readonly css: string };

// words are the controls the sentence marks in braces and relies those it leans on unmarked; first is done before either is read, do is the step itself, then is what shows once it is done, and only marks a step for a tablet or for motion.
export interface Step {
  readonly id: string;
  readonly act: string;
  readonly see: string;
  readonly at: string;
  readonly words: readonly Words[];
  readonly relies?: readonly Words[];
  readonly first?: readonly Act[];
  readonly do?: readonly Act[];
  readonly then?: readonly Words[];
  readonly kind?: "play" | "commission";
  readonly only?: "touch" | "motion";
}

export interface Lesson {
  readonly id: string;
  readonly title: string;
  readonly gloss: string;
  readonly aside?: string;
  readonly steps: readonly Step[];
}

export interface Ladder {
  readonly route: string;
  readonly what: string;
  readonly entered?: string;
  readonly lessons: readonly Lesson[];
}

export function phrasesOf(sentence: string): readonly string[] {
  return [...sentence.matchAll(/\{([^{}]+)\}/g)].map((m) => m[1] ?? "");
}

export const text = (css: string, says: string, match?: "starts" | "contains"): Words =>
  match ? { css, says, read: "text", match } : { css, says, read: "text" };
export const aria = (css: string, says: string): Words => ({ css, says, read: "aria" });
export const road = (css: string, says: string): Words => ({ css, says, read: "road" });
export const option = (css: string, says: string): Words => ({ css, says, read: "option" });
export const labelled = (css: string, says: string): Words => ({ css, says, read: "label" });
export const click = (css: string): Act => ({ kind: "click", css });
export const set = (css: string, value: string): Act => ({ kind: "set", css, value });
export const choose = (css: string, option: string): Act => ({ kind: "choose", css, option });
export const wait = (css: string): Act => ({ kind: "wait", css });
```

`phrasesOf` is the one parser of a sentence's braces: the check uses it now, and Sub 5's page uses it to draw each marked word as its control. The constructors are the ones every room's ladder repeats (the round's `rooms.ts` redefines three of them), so Sub 6 imports them rather than writing eight copies.

```ts
import { NAV_ITEMS, ROUTE_CHILDREN, ROUTE_NAMES, seatOf } from "../../../layouts/nav.ts";
import { EXPLORER } from "./explorer.ts";
import type { Ladder } from "./shape.ts";

export const LADDERS: readonly Ladder[] = [EXPLORER];

export const LEANED_ON: readonly string[] = ["/faq/", "/glossary/"];

const AWAITS = "its ladder lands with Issue #1252";
export const LEFT_UNTAUGHT: Readonly<Record<string, string>> = {
  "/seed-of-the-day/": AWAITS,
  "/writing-room/": AWAITS,
  "/reading-room/": AWAITS,
  "/print-room/": AWAITS,
  "/gallery/": AWAITS,
  "/prospect/": AWAITS,
  "/ribbon/": AWAITS,
  "/explorer/portfolio/": AWAITS,
};

export interface RouteTree {
  readonly routes: readonly string[];
  readonly seat: (route: string) => string | null | undefined;
  readonly names: Readonly<Record<string, string>>;
}

export const ROUTE_TREE: RouteTree = {
  routes: [...new Set([...NAV_ITEMS.map((i) => i.href), ...Object.values(ROUTE_CHILDREN).flat()])],
  seat: seatOf,
  names: ROUTE_NAMES,
};

export function untaught(ladders: readonly Ladder[] = LADDERS, tree: RouteTree = ROUTE_TREE): readonly string[] {
  const covered = new Set([...ladders.map((l) => l.route), ...LEANED_ON, ...Object.keys(LEFT_UNTAUGHT)]);
  return tree.routes.filter((r) => !covered.has(r));
}

export interface Seated {
  readonly ladder: Ladder;
  readonly name: string;
  readonly under: readonly Seated[];
}

const nameIn = (tree: RouteTree, route: string): string => {
  const name = tree.names[route];
  if (name === undefined) throw new Error(`no ROUTE_NAMES entry for ${route}`);
  return name;
};

export function seated(ladders: readonly Ladder[] = LADDERS, tree: RouteTree = ROUTE_TREE): readonly Seated[] {
  return ladders
    .filter((l) => tree.seat(l.route) === null)
    .map((ladder) => ({
      ladder,
      name: nameIn(tree, ladder.route),
      under: ladders
        .filter((l) => tree.seat(l.route) === ladder.route)
        .map((l) => ({ ladder: l, name: nameIn(tree, l.route), under: [] })),
    }));
}
```
