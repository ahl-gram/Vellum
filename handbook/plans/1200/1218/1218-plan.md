**So, in plain words for Alex.** The "keep" is the list of things a caption on the Ribbon must stay clear of: the road, the towns, the compass, the league marks, the summit's mountain. Importing does avoid the drift the issue names, and the measurement shows it. Today, if someone widens the road by one unit and forgets the keep, 44 caption lines on the antique scroll and 39 on the ink scroll end up on the road's drawn edge, and all 66 Ribbon tests stay green. After the fix the keep reads the road's own width, so the same change leaves 0 captions on the road. The test importing matters too. If the keep were ever left with its own copy again, the importing test catches it (on 29 and 32 lines), where the copied test catches nothing. And when the road gets narrower, the copied test raises a false alarm on 29 lines that are fine, while the importing test does not. Importing costs no independence from the keep's own arithmetic: a keep that computes its clearance wrongly is still caught (34 lines, and 11 to 16 lines). What importing does not cover is any change to how the road is drawn other than the width number. Thicker edge dots leave 21 and 28 caption lines on the road's ink, and an edge drawn further out leaves 25 and 9. In both cases every test stays green, before the fix and after it, whether the test copies or imports. The same gap is wider at the summit and at the ink scroll's fifth-league numbers, because no test looks at them at all. Remove the keep's mountain box and 65 to 66 lines sit on a mountain with every test green, and this pull request moves exactly that code. Alex ruled (comment 6619920056) to build as filed and carry the checks in a follow-up issue (written out below), so this pull request leaves that gap open and says so in its body.

import { textBox, WIDTH_FACTOR, type Box } from "../src/render/geometry.ts";
import { attrsOf } from "./dress-svg.ts";

export type RoadInk = {
  readonly kind: "casing" | "span";
  readonly halfStroke: number;
  readonly pts: ReadonlyArray<readonly [number, number]>;
};

export function roadInkOf(svg: string): RoadInk[] {
  const out: RoadInk[] = [];
  for (const m of svg.matchAll(/<path ([^>]*?)\/>/g)) {
    const a = attrsOf(m[1] as string);
    const dash = a["stroke-dasharray"];
    if (a.class !== "road" || dash === "0.2 3") continue;
    const nums = ((a.d ?? "").match(/-?\d+(?:\.\d+)?/g) ?? []).map(Number);
    const pts: Array<readonly [number, number]> = [];
    for (let i = 0; i + 1 < nums.length; i += 2) pts.push([nums[i] as number, nums[i + 1] as number]);
    out.push({ kind: dash === undefined ? "span" : "casing", halfStroke: Number(a["stroke-width"]) / 2, pts });
  }
  return out;
}

const pointGap = (b: Box, x: number, y: number): number =>
  Math.hypot(Math.max(b.x - x, 0, x - (b.x + b.w)), Math.max(b.y - y, 0, y - (b.y + b.h)));

function pointSegment(px: number, py: number, a: readonly [number, number], c: readonly [number, number]): number {
  const [dx, dy] = [c[0] - a[0], c[1] - a[1]];
  const len2 = dx * dx + dy * dy;
  const t = len2 === 0 ? 0 : Math.max(0, Math.min(1, ((px - a[0]) * dx + (py - a[1]) * dy) / len2));
  return Math.hypot(px - (a[0] + t * dx), py - (a[1] + t * dy));
}

function crosses(b: Box, a: readonly [number, number], c: readonly [number, number]): boolean {
  let [t0, t1] = [0, 1];
  const [dx, dy] = [c[0] - a[0], c[1] - a[1]];
  const sides: ReadonlyArray<readonly [number, number]> = [
    [-dx, a[0] - b.x],
    [dx, b.x + b.w - a[0]],
    [-dy, a[1] - b.y],
    [dy, b.y + b.h - a[1]],
  ];
  for (const [p, q] of sides) {
    if (p === 0) {
      if (q < 0) return false;
      continue;
    }
    if (p < 0) t0 = Math.max(t0, q / p);
    else t1 = Math.min(t1, q / p);
    if (t0 > t1) return false;
  }
  return true;
}

export function inkGap(b: Box, ink: RoadInk): number {
  const corners = [
    [b.x, b.y],
    [b.x + b.w, b.y],
    [b.x, b.y + b.h],
    [b.x + b.w, b.y + b.h],
  ] as const;
  let best = Infinity;
  for (let i = 1; i < ink.pts.length; i++) {
    const [a, c] = [ink.pts[i - 1], ink.pts[i]] as [readonly [number, number], readonly [number, number]];
    if (crosses(b, a, c)) return -ink.halfStroke;
    best = Math.min(best, pointGap(b, ...a), pointGap(b, ...c), ...corners.map(([x, y]) => pointSegment(x, y, a, c)));
  }
  return best - ink.halfStroke;
}

export function summitMountainsOf(svg: string): Box[] {
  const drawn = /<g transform="translate\((-?[\d.]+) (-?[\d.]+)\) scale\(1\.05\)"><path d="(M-9 0L-3\.4 -10[^"]*)"/g;
  return [...svg.matchAll(drawn)].map((m) => {
    const nums = ((m[3] as string).match(/-?\d+(?:\.\d+)?/g) ?? []).map(Number);
    const xs = nums.filter((_, i) => i % 2 === 0);
    const ys = nums.filter((_, i) => i % 2 === 1);
    const [x, y, s] = [Number(m[1]), Number(m[2]), 1.05];
    const [x0, y0] = [Math.min(...xs), Math.min(...ys)];
    return { x: x + x0 * s, y: y + y0 * s, w: (Math.max(...xs) - x0) * s, h: (Math.max(...ys) - y0) * s };
  });
}

export function fifthNumeralsOf(svg: string): Box[] {
  return [...svg.matchAll(/<text ([^>]*)>(\d+)<\/text>/g)].flatMap((m) => {
    const a = attrsOf(m[1] as string);
    if (a["font-size"] !== "7" || a["text-anchor"] !== undefined) return [];
    return [textBox(Number(a.x), Number(a.y), m[2] as string, 7, "start", WIDTH_FACTOR.mixed)];
  });
}

export const separation = (a: Box, b: Box): number =>
  Math.max(a.x - (b.x + b.w), b.x - (a.x + a.w), a.y - (b.y + b.h), b.y - (a.y + a.h));
