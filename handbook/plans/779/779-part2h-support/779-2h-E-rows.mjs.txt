const count = (t, f) => t.split(f).length - 1;
let bad = 0;
const mode = process.argv[2] ?? "count";
for (const [id, edits] of Object.entries(ROWS)) {
  // Apply in order to a working copy per file, so each find is counted where it is applied.
  const work = {};
  for (const e of edits) {
    if (!(e.file in work)) work[e.file] = readFileSync(ROOT + e.file, "utf8");
    const orig = count(readFileSync(ROOT + e.file, "utf8"), e.find);
    const n = count(work[e.file], e.find);
    if (n !== 1 || orig !== 1) bad++;
    if (mode === "count") console.log(`${id}\t${e.file}\torig=${orig}\tapplied=${n}`);
    work[e.file] = work[e.file].replace(e.find, () => e.replace);
  }
}
if (mode === "count") console.log("rows", Object.keys(ROWS).length, "bad", bad);
if (mode === "json") for (const [id, edits] of Object.entries(ROWS)) console.log(`${id}\t${JSON.stringify(edits)}`);
