// 779-2h-E: measure each row's unit reds on a SCRATCH copy of the files the source-reading tests read (nothing under the repo is written).
import { cpSync, mkdirSync, readFileSync, rmSync, symlinkSync, writeFileSync, existsSync } from "node:fs";
import { spawnSync } from "node:child_process";
import { dirname, join } from "node:path";
import { ROWS } from "./779-2h-E-rows.mjs";

const REPO = "<repo>/";
const TREE = "<scratchpad>/779-2h-E-tree/";
const FILES = {
  "broadside": "test/site/broadside.test.ts",
  "explorer-room": "test/site/explorer-room.test.ts",
  "explorer-page": "test/site/explorer-page.test.ts",
  "table-address": "test/site/table-address.test.ts",
  "tip-affordance": "test/site/tip-affordance.test.ts",
  "atelier-kit": "test/site/atelier-kit.test.ts",
  "living-chart-css": "test/site/living-chart-css.test.ts",
  "house-style": "test/site/house-style.test.ts",
  "app-bundles": "test/site/app-bundles.test.ts",
  "rdw": "test/site/region-detail-wiring.test.ts",
};

function build() {
  rmSync(TREE, { recursive: true, force: true });
  mkdirSync(TREE, { recursive: true });
  for (const d of ["src", "test-support", "scripts", "test/site", "public"]) cpSync(REPO + d, TREE + d, { recursive: true });
  cpSync(REPO + "package.json", TREE + "package.json");
  symlinkSync("<repo>/node_modules", TREE + "node_modules");
}

function run(files) {
  const r = spawnSync(process.execPath, ["--test", "--test-reporter=tap", ...files], { cwd: TREE, encoding: "utf8", timeout: 600000 });
  const out = r.stdout || "";
  const reds = [];
  let file = null;
  for (const line of out.split("\n")) {
    const m = /^not ok \d+ - (.*)$/.exec(line);
    if (m) reds.push(m[1].trim());
  }
  return reds;
}

const which = process.argv.slice(2);
build();
const groups = (process.env.GROUPS ?? "broadside,explorer-room,explorer-page,table-address,tip-affordance,atelier-kit,living-chart-css,house-style,app-bundles").split(",");
const files = groups.map((g) => FILES[g]);
const base = new Set(run(files));
console.log("BASELINE reds:", JSON.stringify([...base]));
for (const [id, edits] of Object.entries(ROWS)) {
  if (which.length && !which.includes(id)) continue;
  const touched = new Set();
  for (const e of edits) {
    const p = TREE + e.file;
    const t = readFileSync(p, "utf8");
    writeFileSync(p, t.replace(e.find, () => e.replace));
    touched.add(e.file);
  }
  const reds = run(files).filter((n) => !base.has(n));
  console.log(`${id}\t${JSON.stringify(reds)}`);
  for (const f of touched) cpSync(REPO + f, TREE + f);
}
